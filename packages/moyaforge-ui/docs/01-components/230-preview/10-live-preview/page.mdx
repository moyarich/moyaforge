# Counter with source code

This example uses the standard toolbar and adds a short instruction in the footer.
Slots can be omitted, replaced or reordered without modifying Preview.

:::code-preview[Counter with source]{react replacements="portable"}

import {
  Preview,
  PreviewHeader,
  PreviewTitle,
  PreviewActions,
  PreviewCodeToggle,
  PreviewCopyButton,
  PreviewContent,
  PreviewSource,
  PreviewFooter,
} from "../../../../src/components/Preview/index.js";

import "../../counterExample.css";

import CounterExample from "../../CounterExample.js";
import counterSource from "../../CounterExample.tsx?raw";

export default function LivePreviewExample() {
return (

<Preview source={counterSource}>
  <PreviewHeader>
    <PreviewTitle>Interactive counter</PreviewTitle>
    <PreviewActions>
      <PreviewCodeToggle />
      <PreviewCopyButton />
    </PreviewActions>
  </PreviewHeader>
  <PreviewContent>
    <CounterExample />
  </PreviewContent>
  <PreviewSource code={counterSource} language="tsx" />
  <PreviewFooter>Click the button, then inspect its source.</PreviewFooter>
</Preview>
);
}

:::

## Customize the Code button

Use `asChild` to give the toggle your own button label or styling. The button still controls the same source panel.

```tsx
<PreviewCodeToggle asChild>
  <button type="button" className="my-code-trigger">
    View source
  </button>
</PreviewCodeToggle>
```
